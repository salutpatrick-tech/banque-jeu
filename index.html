const functions = require("firebase-functions"); const admin = require("firebase-admin"); admin.initializeApp(); const db = admin.firestore(); /** * Middleware / Vérification des droits administrateur */ const verifyAdmin = async (context) => { if (!context.auth) { throw new functions.https.HttpsError( "unauthenticated", "L'utilisateur doit être authentifié." ); } if (context.auth.token.admin !== true) { throw new functions.https.HttpsError( "permission-denied", "Seuls les administrateurs peuvent effectuer cette action." ); } }; /** * 1. Annuler une transaction (Callable Function) */ exports.cancelTransaction = functions.https.onCall(async (data, context) => { await verifyAdmin(context); const { transactionId, reason } = data; if (!transactionId) { throw new functions.https.HttpsError( "invalid-argument", "L'ID de transaction est requis." ); } const transactionRef = db.collection("transactions").doc(transactionId); return db.runTransaction(async (t) => { const doc = await t.get(transactionRef); if (!doc.exists) { throw new functions.https.HttpsError("not-found", "Transaction introuvable."); } const txData = doc.data(); if (txData.status === "cancelled") { throw new functions.https.HttpsError( "failed-precondition", "Cette transaction est déjà annulée." ); } // Mise à jour du statut et ajout à l'historique t.update(transactionRef, { status: "cancelled", updatedAt: admin.firestore.FieldValue.serverTimestamp(), updatedBy: context.auth.uid, history: admin.firestore.FieldValue.arrayUnion({ action: "cancel", previousAmount: txData.amount, previousStatus: txData.status, reason: reason || "Annulation administrative", performedBy: context.auth.uid, timestamp: new Date().toISOString() }) }); return { success: true, message: "Transaction annulée avec succès." }; }); }); /** * 2. Modifier le montant d'une transaction (Callable Function) */ exports.updateTransaction = functions.https.onCall(async (data, context) => { await verifyAdmin(context); const { transactionId, newAmount, reason } = data; if (!transactionId || typeof newAmount !== "number") { throw new functions.https.HttpsError( "invalid-argument", "Paramètres invalides (transactionId et newAmount requis)." ); } const transactionRef = db.collection("transactions").doc(transactionId); return db.runTransaction(async (t) => { const doc = await t.get(transactionRef); if (!doc.exists) { throw new functions.https.HttpsError("not-found", "Transaction introuvable."); } const txData = doc.data(); if (txData.status === "cancelled") { throw new functions.https.HttpsError( "failed-precondition", "Impossible de modifier une transaction annulée." ); } const oldAmount = txData.amount; t.update(transactionRef, { amount: newAmount, status: "modified", updatedAt: admin.firestore.FieldValue.serverTimestamp(), updatedBy: context.auth.uid, history: admin.firestore.FieldValue.arrayUnion({ action: "update", oldAmount: oldAmount, newAmount: newAmount, reason: reason || "Modification administrative", performedBy: context.auth.uid, timestamp: new Date().toISOString() }) }); return { success: true, message: "Transaction modifiée avec succès." }; }); }); /** * 3. Déclencheur Firestore (Trigger) * Réajuste automatiquement le solde du compte utilisateur lors d'une annulation ou modification. */ exports.onTransactionUpdate = functions.firestore .document("transactions/{transactionId}") .onUpdate(async (change, context) => { const beforeData = change.before.data(); const afterData = change.after.data(); // Si le montant ou le statut n'a pas changé, rien à recalculer if ( beforeData.amount === afterData.amount && beforeData.status === afterData.status ) { return null; } const userId = afterData.userId; if (!userId) return null; const userRef = db.collection("users").doc(userId); return db.runTransaction(async (t) => { const userDoc = await t.get(userRef); if (!userDoc.exists) return; let balanceAdjustment = 0; // Cas 1 : La transaction vient d'être annulée -> On annule l'impact de l'ancien montant if (beforeData.status !== "cancelled" && afterData.status === "cancelled") { balanceAdjustment = -beforeData.amount; } // Cas 2 : Le montant a été modifié (sans être annulé) else if (afterData.status !== "cancelled") { balanceAdjustment = afterData.amount - beforeData.amount; } if (balanceAdjustment !== 0) { t.update(userRef, { balance: admin.firestore.FieldValue.increment(balanceAdjustment) }); } }); });
