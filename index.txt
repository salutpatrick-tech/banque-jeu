    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js";
        import { getDatabase, ref, set, push, onValue, remove } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-database.js";

        // Configuration Firebase
        const firebaseConfig = {
            apiKey: "VOTRE_API_KEY",
            authDomain: "VOTRE_PROJECT_ID.firebaseapp.com",
            databaseURL: "https://VOTRE_PROJECT_ID-default-rtdb.firebaseio.com",
            projectId: "VOTRE_PROJECT_ID",
            storageBucket: "VOTRE_PROJECT_ID.appspot.com",
            messagingSenderId: "VOTRE_SENDER_ID",
            appId: "VOTRE_APP_ID"
        };

        const app = initializeApp(firebaseConfig);
        const db = getDatabase(app);

        // Références Firebase
        const joueursRef = ref(db, 'joueurs');
        const histoRef = ref(db, 'historique');
        const demandesRef = ref(db, 'demandes');
        const chronoRef = ref(db, 'config/chrono');
        const derniereActiviteRef = ref(db, 'config/derniereActivite');
        const alerteRef = ref(db, 'config/alerteJokerActive');

        // Variables globales locales
        window.localJoueurs = {};
        window.isAdmin = false;
        let chronoInterval = null;

        // -------------------------------------------------------------
        // NAVIGATION PAR ONGLETS
        // -------------------------------------------------------------
        window.changerOnglet = function(tabId, element) {
            document.querySelectorAll('.tab-content').forEach(tab => tab.classList.remove('active'));
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));

            document.getElementById(`tab-${tabId}`).classList.add('active');
            element.classList.add('active');
        };

        // -------------------------------------------------------------
        // MODE ADMINISTRATEUR
        // -------------------------------------------------------------
        window.connexionAdmin = function() {
            const pin = document.getElementById('admin-pin').value;
            if (pin === "1234") { // Code Admin par défaut
                window.isAdmin = true;
                document.getElementById('admin-badge').textContent = "🔓 Mode Admin";
                document.getElementById('admin-badge').classList.add('active');
                document.getElementById('admin-auth-box').style.display = 'none';
                document.querySelectorAll('.admin-only').forEach(el => el.style.display = 'block');
                document.getElementById('admin-pin').value = '';
                actualiserAffichageJoueurs();
            } else {
                alert("Code Administrateur incorrect.");
            }
        };

        window.deconnexionAdmin = function() {
            window.isAdmin = false;
            document.getElementById('admin-badge').textContent = "🔒 Mode Joueur";
            document.getElementById('admin-badge').classList.remove('active');
            document.getElementById('admin-auth-box').style.display = 'block';
            document.querySelectorAll('.admin-only').forEach(el => el.style.display = 'none');
            actualiserAffichageJoueurs();
        };

        // -------------------------------------------------------------
        // SYNCHRONISATION BANQUE & JOUEURS
        // -------------------------------------------------------------
        onValue(joueursRef, (snapshot) => {
            window.localJoueurs = snapshot.val() || {};
            actualiserAffichageJoueurs();
        });

        function actualiserAffichageJoueurs() {
            const data = window.localJoueurs;
            const listeUi = document.getElementById('liste-joueurs');
            const selectTransac = document.getElementById('transac-joueur');
            const selectEmetteur = document.getElementById('demande-emetteur');
            const selectDest = document.getElementById('demande-destinataire');

            listeUi.innerHTML = '';
            selectTransac.innerHTML = '<option value="">Sélectionner un joueur</option>';
            selectEmetteur.innerHTML = '<option value="">Sélectionner votre nom</option>';
            selectDest.innerHTML = '<option value="BANQUE">🏦 La Banque</option>';

            if (Object.keys(data).length === 0) {
                listeUi.innerHTML = '<li>Aucun joueur enregistré.</li>';
                return;
            }

            Object.entries(data).forEach(([id, j]) => {
                // Remplissage de la liste
                const li = document.createElement('li');
                li.innerHTML = `
                    <div>
                        <strong>${j.nom}</strong>
                        <div class="solde">${j.solde || 0} €</div>
                        <div class="badge-joker">🃏 Jokers : ${j.jokers || 0} / 2</div>
                    </div>
                    <div>
                        ${window.isAdmin ? `<button class="btn-danger-sm" onclick="supprimerJoueur('${id}')">❌</button>` : ''}
                    </div>
                `;
                listeUi.appendChild(li);

                // Option des menus déroulants
                const opt1 = new Option(j.nom, id);
                const opt2 = new Option(j.nom, id);
                const opt3 = new Option(j.nom, id);
                selectTransac.add(opt1);
                selectEmetteur.add(opt2);
                selectDest.add(opt3);
            });
        }

        // -------------------------------------------------------------
        // ACTIONS ET TRANSACTIONS
        // -------------------------------------------------------------
        function verifierPin(joueurId) {
            const joueur = window.localJoueurs[joueurId];
            if (!joueur) return false;
            const pinSaisi = prompt(`Code PIN de ${joueur.nom} :`);
            return pinSaisi === joueur.pin;
        }

        function enregistrerActivite() {
            set(derniereActiviteRef, Date.now());
            set(alerteRef, false);
        }

        window.creerJoueur = function() {
            const nom = document.getElementById('nouveau-nom').value.trim();
            const pin = document.getElementById('nouveau-pin').value.trim();

            if (!nom || pin.length !== 4) {
                alert("Nom requis et PIN de 4 chiffres.");
                return;
            }

            const newRef = push(joueursRef);
            set(newRef, { nom, pin, solde: 100, jokers: 0 });

            push(histoRef, {
                texte: `👤 Nouveau joueur créé : ${nom}`,
                date: new Date().toLocaleTimeString('fr-FR')
            });

            document.getElementById('nouveau-nom').value = '';
            document.getElementById('nouveau-pin').value = '';
            enregistrerActivite();
        };

        window.supprimerJoueur = function(id) {
            if (confirm("Voulez-vous vraiment supprimer ce joueur ?")) {
                remove(ref(db, `joueurs/${id}`));
            }
        };

        window.validerTransaction = function() {
            const id = document.getElementById('transac-joueur').value;
            const type = document.getElementById('transac-type').value;
            const montant = parseInt(document.getElementById('transac-montant').value);

            if (!id) return alert("Sélectionnez un joueur.");
            if (!verifierPin(id)) return alert("Code PIN incorrect !");

            const joueur = window.localJoueurs[id];
            let soldeActuel = joueur.solde || 0;
            let jokersActuels = joueur.jokers || 0;

            if (type === 'depot') {
                if (isNaN(montant) || montant <= 0) return alert("Montant invalide.");
                set(ref(db, `joueurs/${id}/solde`), soldeActuel + montant);
                push(histoRef, { texte: `💳 ${joueur.nom} a déposé ${montant} €`, date: new Date().toLocaleTimeString('fr-FR') });
            } else if (type === 'retrait') {
                if (isNaN(montant) || montant <= 0) return alert("Montant invalide.");
                set(ref(db, `joueurs/${id}/solde`), soldeActuel - montant);
                push(histoRef, { texte: `💳 ${joueur.nom} a retiré ${montant} €`, date: new Date().toLocaleTimeString('fr-FR') });
            } else if (type === 'joker') {
                if (jokersActuels <= 0) return alert(`${joueur.nom} n'a aucun Joker !`);
                set(ref(db, `joueurs/${id}/jokers`), jokersActuels - 1);
                push(histoRef, { texte: `🃏 ${joueur.nom} a utilisé 1 Joker (${jokersActuels - 1}/2 restant)`, date: new Date().toLocaleTimeString('fr-FR') });
            }

            document.getElementById('transac-montant').value = '';
            enregistrerActivite();
        };

        // -------------------------------------------------------------
        // JOKER MYSTÈRE & INACTIVITÉ
        // -------------------------------------------------------------
        onValue(alerteRef, (snapshot) => {
            const active = snapshot.val();
            const box = document.getElementById('alerte-joker-surprise');
            if (box) box.style.display = active ? 'block' : 'none';
        });

        window.reclamerJokerInactivite = function() {
            const liste = Object.entries(window.localJoueurs);
            if (liste.length === 0) return;

            let text = "Qui réclame ce Joker ?\n";
            liste.forEach(([id, j], idx) => text += `${idx + 1}. ${j.nom}\n`);
            const choix = parseInt(prompt(text)) - 1;

            if (isNaN(choix) || choix < 0 || choix >= liste.length) return alert("Choix invalide.");

            const [joueurId, joueur] = liste[choix];
            if (!verifierPin(joueurId)) return alert("Code PIN incorrect !");

            const jokersActuels = joueur.jokers || 0;

            if (jokersActuels >= 2) {
                alert(`⚠️ ${joueur.nom}, tu possèdes déjà le maximum de 2 Jokers !\nCe 3ᵉ Joker doit être consommé immédiatement !`);
                push(histoRef, {
                    texte: `⚡ JOKER CONSOMMÉ : ${joueur.nom} a réclamé le Joker mystère mais avait déjà 2 Jokers. Il l'a utilisé sur-le-champ !`,
                    date: new Date().toLocaleTimeString('fr-FR')
                });
            } else {
                const nouveauTotal = jokersActuels + 1;
                set(ref(db, `joueurs/${joueurId}/jokers`), nouveauTotal);
                push(histoRef, {
                    texte: `⚡ JOKER MYSTÈRE : ${joueur.nom} a réclamé le Joker mystère d'inactivité ! (${nouveauTotal}/2)`,
                    date: new Date().toLocaleTimeString('fr-FR')
                });
                alert(`Bravo ${joueur.nom} ! Joker ajouté à ton stock (${nouveauTotal}/2).`);
            }

            set(alerteRef, false);
            set(derniereActiviteRef, Date.now());
        };

        // -------------------------------------------------------------
        // DEMANDES ET VIREMENTS
        // -------------------------------------------------------------
        window.soumettreDemande = function() {
            const emetteurId = document.getElementById('demande-emetteur').value;
            const destId = document.getElementById('demande-destinataire').value;
            const type = document.getElementById('demande-type').value;
            const montant = parseInt(document.getElementById('demande-montant').value) || 0;
            const note = document.getElementById('demande-note').value.trim();

            if (!emetteurId) return alert("Sélectionnez votre nom.");
            if (!verifierPin(emetteurId)) return alert("Code PIN incorrect !");

            const emetteurNom = window.localJoueurs[emetteurId].nom;
            const destNom = destId === "BANQUE" ? "la Banque" : window.localJoueurs[destId].nom;

            const newDemandeRef = push(demandesRef);
            set(newDemandeRef, {
                emetteurId, emetteurNom, destId, destNom, type, montant, note, status: 'attente'
            });

            push(histoRef, {
                texte: `📩 Demande de ${emetteurNom} vers ${destNom} (${type} : ${montant} €)`,
                date: new Date().toLocaleTimeString('fr-FR')
            });

            document.getElementById('demande-montant').value = '';
            document.getElementById('demande-note').value = '';
            enregistrerActivite();
        };

        onValue(demandesRef, (snapshot) => {
            const data = snapshot.val() || {};
            const conteneur = document.getElementById('liste-demandes');
            conteneur.innerHTML = '';

            if (Object.keys(data).length === 0) {
                conteneur.innerHTML = '<p>Aucune demande active.</p>';
                return;
            }

            Object.entries(data).forEach(([id, d]) => {
                const item = document.createElement('div');
                item.className = 'demande-item';
                item.innerHTML = `
                    <div><strong>${d.emetteurNom}</strong> ➔ <strong>${d.destNom}</strong> (${d.type})</div>
                    <div>Montant : <strong>${d.montant} €</strong> ${d.note ? `| Note: <em>${d.note}</em>` : ''}</div>
                    <div class="demande-actions">
                        <button class="btn-success" onclick="traiterDemande('${id}', true)">Accepter</button>
                        <button class="btn-danger" onclick="traiterDemande('${id}', false)">Refuser</button>
                    </div>
                `;
                conteneur.appendChild(item);
            });
        });

        window.traiterDemande = function(id, accepter) {
            if (accepter) {
                push(histoRef, { texte: `✅ Demande validée`, date: new Date().toLocaleTimeString('fr-FR') });
            } else {
                push(histoRef, { texte: `❌ Demande refusée`, date: new Date().toLocaleTimeString('fr-FR') });
            }
            remove(ref(db, `demandes/${id}`));
            enregistrerActivite();
        };

        // -------------------------------------------------------------
        // HISTORIQUE & CHRONOMÈTRE
        // -------------------------------------------------------------
        onValue(histoRef, (snapshot) => {
            const data = snapshot.val() || {};
            const conteneur = document.getElementById('historique-liste');
            conteneur.innerHTML = '';

            Object.values(data).reverse().forEach(h => {
                const li = document.createElement('li');
                li.innerHTML = `<span>${h.texte}</span><small style="color: #64748b">${h.date}</small>`;
                conteneur.appendChild(li);
            });
        });

        onValue(chronoRef, (snapshot) => {
            const chrono = snapshot.val();
            const banner = document.getElementById('chrono-banner');

            if (!chrono || !chrono.actif) {
                banner.style.display = 'none';
                if (chronoInterval) clearInterval(chronoInterval);
                return;
            }

            banner.style.display = 'block';
            document.getElementById('chrono-description').textContent = `⏱️ ${chrono.description}`;

            if (chronoInterval) clearInterval(chronoInterval);

            chronoInterval = setInterval(() => {
                const diff = chrono.fin - Date.now();
                if (diff <= 0) {
                    clearInterval(chronoInterval);
                    banner.style.display = 'none';
                    return;
                }

                document.getElementById('t-jours').textContent = Math.floor(diff / (1000 * 60 * 60 * 24));
                document.getElementById('t-heures').textContent = Math.floor((diff / (1000 * 60 * 60)) % 24).toString().padStart(2, '0');
                document.getElementById('t-minutes').textContent = Math.floor((diff / 1000 / 60) % 60).toString().padStart(2, '0');
                document.getElementById('t-secondes').textContent = Math.floor((diff / 1000) % 60).toString().padStart(2, '0');
            }, 1000);
        });

        window.lancerChrono = function() {
            const desc = document.getElementById('chrono-desc').value.trim() || "Défi en cours";
            const dateVal = document.getElementById('chrono-date').value;

            if (!dateVal) return alert("Sélectionnez une date de fin.");

            const fin = new Date(dateVal).getTime();
            set(chronoRef, { actif: true, description: desc, fin });
        };

        window.arreterChrono = function() {
            set(chronoRef, { actif: false });
        };
    </script>
